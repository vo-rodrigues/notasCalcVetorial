\section{Espaços métricos}

Na Proposição~\ref{prop:distanciaRn}, vimos que a distância euclidiana usual de $\mathbb R^n$ satisfaz algumas propriedades básicas típicas do que se espera que uma noção de distância possua.
O conceito de espaço métrico consiste em nada mais do que um conjunto $M$ munido de uma função $d$ que satisfaz exatamente aquelas propriedades.

\begin{definition}\index{espaço métrico}\index{métrica}\label{def:espMetrico}
    Seja $M$ um conjunto.
    Uma \emph{métrica} em $M$ é uma função $d:M\times M\to \mathbb R$ que satisfaz, para todos $p,q,r\in M$, as seguintes propriedades:
    \begin{enumerate}[label=(\roman*)]
        \item $d(p,q)\geq 0$, e $d(p,q)=0$ se, e somente se, $p=q$ \quad (positividade e separação de pontos);
        \item $d(p,q)=d(q,p)$ \quad (simetria);
        \item $d(p,r)\leq d(p,q)+d(q,r)$ \quad (desigualdade triangular).
    \end{enumerate}
    Um \emph{espaço métrico} é um par $(M,d)$, em que $M$ é um conjunto e $d$ é uma métrica em $M$.
\end{definition}

\begin{example}
    Quando $M=\mathbb R^n$ e $d$ é a distância euclidiana usual, obtemos o principal exemplo deste texto.
    Nesse caso, para $p,q\in\mathbb R^n$, temos
    \[
        d(p,q)=\|p-q\|.
    \]
\end{example}

A seguir, definiremos generalizações das noções de intervalo aberto e intervalo fechado de $\mathbb R$.
Essas noções serão essenciais para o futuro estudo de limites, continuidade e diferenciabilidade.

\begin{definition}\index{bola}\index{bola aberta}\index{bola fechada}
    Sejam $(M,d)$ um espaço métrico, $p \in M$ e $r>0$.
    A \emph{bola aberta} de centro $p$ e raio $r$ é o conjunto:
    \begin{equation*}
        B_M(p, r) = \{q \in M : d(p, q) < r\}.
    \end{equation*}
    A \emph{bola fechada} de centro $p$ e raio $r$ é o conjunto:
    \begin{equation*}
        B_M[p, r] = \{q \in M : d(p, q) \leq r\}.
    \end{equation*}
    Quando o espaço métrico estiver claro pelo contexto, escreveremos simplesmente $B(p,r)$ e $B[p,r]$.
\end{definition}

\begin{example}
    No caso em que $M=\mathbb R^n$ e $d$ é a distância euclidiana usual, temos
    \[
        B_{\mathbb R^n}(p,r)=\{q\in\mathbb R^n:\|q-p\|<r\}
    \]
    e
    \[
        B_{\mathbb R^n}[p,r]=\{q\in\mathbb R^n:\|q-p\|\leq r\}.
    \]

    No caso particular de $\mathbb R^2$, a bola aberta $B(p,r)$ é o disco de centro $p$ e raio $r$, sem incluir a circunferência que o delimita.
    Na Figura~\ref{fig:bolaAbertaR2}, essa circunferência aparece tracejada apenas para indicar o contorno da bola.

    \subimport{}{espacosMetricos-figuraBolaAbertaR2}

    Em $\mathbb R$, a bola aberta de centro $p$ e raio $r$ é exatamente o intervalo aberto $]p-r, p+r[$, como ilustrado na Figura~\ref{fig:bolaAbertaR}.
    Analogamente, a bola fechada $B_{\mathbb R}[p,r]$ é o intervalo fechado $[p-r,p+r]$.

    \subimport{}{espacosMetricos-figuraBolaAbertaR}
\end{example}


Intuitivamente, qualquer subconjunto de um espaço métrico também é, com a métrica herdada, um espaço métrico.
Isso é formalizado na seguinte definição.

\begin{definition}\index{espaço métrico!subespaço}
    Sejam $(M,d)$ um espaço métrico e $A\subseteq M$ um subconjunto de $M$.
    O \emph{subespaço} de $M$ determinado por $A$ é o par $(A, d|_{A\times A})$, onde $d|_{A\times A}$ é a restrição de $d$ ao produto cartesiano $A\times A$.
\end{definition}

É imediato da Definição~\ref{def:espMetrico} que o subespaço de um espaço métrico é, de fato, um espaço métrico (veja o Exercício~\ref{ex:subespacoEspacoMetrico}).

As bolas abertas do subespaço são obtidas pela interseção das bolas abertas originais com o subespaço.

\begin{proposition}
    Sejam $(M,d)$ um espaço métrico, $A\subseteq M$ e $p\in A$.
    Então, para todo $r>0$, temos
    \[
        B_A(p,r)=B_M(p,r)\cap A.
    \]
\end{proposition}
\begin{proof}
    Dado $q \in B_A(p,r)$, temos $q\in A$ e $d|_{A\times A}(p,q)<r$.
    Como $d|_{A\times A}(p,q)=d(p,q)$, segue que $q \in B_M(p,r)\cap A$.

    Reciprocamente, se $q \in B_M(p,r)\cap A$, então $q\in A$ e $d(p,q)<r$.
    Como $p,q\in A$, temos $d|_{A\times A}(p,q)=d(p,q)<r$, de modo que $q \in B_A(p,r)$.
\end{proof}

\begin{example}
    Considere $A=[0,+\infty[\subseteq\mathbb R$, com a métrica herdada da distância euclidiana usual de $\mathbb R$.
    Então, no subespaço $A$, temos
    \[
        B_A(0,1)
        =
        B_{\mathbb R}(0,1)\cap A
        =
        ]-1,1[\cap [0,+\infty[
        =
        [0,1[.
    \]
    Assim, $[0,1[$ é uma bola aberta do subespaço $A$, embora não seja uma bola aberta de $\mathbb R$.
\end{example}

\begin{example}
    Considere o subespaço
    \[
        A=\mathbb R\times [0,+\infty[=\{(x,y)\in\mathbb R^2:y\geq 0\}\subseteq\mathbb R^2,
    \]
    com a métrica herdada da distância euclidiana usual de $\mathbb R^2$.
    Seja $p=(0,0)$.
    Pela proposição anterior, temos
    \[
        B_A(p,1)=B_{\mathbb R^2}(p,1)\cap A=\{q\in A: d(q, p)<1\}.
    \]

    Segue que
    \[
        B_A((0,0),1)
        =
        \{(x,y)\in\mathbb R^2:x^2+y^2<1 \text{ e } y\geq 0\}.
    \]
    
    Assim, $B_A((0,0),1)$ é o semidisco superior de raio $1$ centrado na origem, como na Figura~\ref{fig:bolaAbertaSubespaco}.
    Observe que o segmento $]-1,1[\times\{0\}$ pertence a essa bola, pois seus pontos pertencem ao subespaço $A$ e distam menos que $1$ de $p$.
    Por outro lado, o arco tracejado não pertence à bola, pois seus pontos distam exatamente $1$ de $p$.
\end{example}

\section*{Exercícios}

\begin{exercise}
    Seja $(M,d)$ um espaço métrico, $p\in M$ e $r>0$.
    Mostre que
    \[
        p\in B(p,r)\subseteq B[p,r].
    \]
\end{exercise}

\begin{exercise}
    Seja $(M,d)$ um espaço métrico, $p\in M$ e sejam $0<r\leq s$ números reais.
    Mostre que
    \[
        B(p,r)\subseteq B(p,s)
        \quad\text{e}\quad
        B[p,r]\subseteq B[p,s].
    \]
\end{exercise}

\begin{exercise}
    Em $\mathbb R$, com a distância euclidiana usual, descreva explicitamente os seguintes conjuntos:
    \[
        B(2,3),\qquad B[2,3],\qquad B(-1,1/2),\qquad B[-1,1/2].
    \]
\end{exercise}

\begin{exercise}
    Em $\mathbb R^2$, com a distância euclidiana usual, descreva explicitamente os conjuntos
    \[
        B((0,0),1),\qquad B[(1,2),1].
    \]
\end{exercise}

\begin{exercise}
    Em $\mathbb R^2$, com a distância euclidiana usual, seja $p=(1,-1)$.
    Determine se os pontos
    \[
        (1,0),\qquad (3,-1),\qquad (2,0),\qquad (1+\sqrt2,-1)
    \]
    pertencem a $B(p,2)$ e a $B[p,2]$.
\end{exercise}

\begin{exercise}\label{ex:subespacoEspacoMetrico}
    Seja $(M,d)$ um espaço métrico e seja $A\subseteq M$.
    Mostre diretamente, a partir da Definição~\ref{def:espMetrico}, que $(A,d|_{A\times A})$ é um espaço métrico.
\end{exercise}

\begin{exercise}
    Considere $A=[0,+\infty[\subseteq\mathbb R$, com a métrica herdada da distância euclidiana usual de $\mathbb R$.
    Calcule:
    \[
        B_A(0,1),\qquad B_A[0,1],\qquad B_A(2,1),\qquad B_A[2,1].
    \]
\end{exercise}

\begin{exercise}
    Considere $A=[0,+\infty[\subseteq\mathbb R$, com a métrica herdada da distância euclidiana usual de $\mathbb R$.
    Mostre que, para todo $r>0$, temos
    \[
        B_A(0,r)=[0,r[.
    \]
\end{exercise}

\begin{exercise}
    Sejam $(M,d)$ um espaço métrico, $A\subseteq M$ e $p\in A$.
    Mostre que, para todo $r>0$, temos a seguinte relação envolvendo bolas fechadas:
    \[
        B_A[p,r]=B_M[p,r]\cap A.
    \]
\end{exercise}

\begin{exercise}
    Considere
    \[
        A=\mathbb R\times[0,+\infty[\subseteq\mathbb R^2
    \]
    com a métrica herdada da distância euclidiana usual de $\mathbb R^2$.
    Calcule explicitamente:
    \[
        B_A((1,0),1)
        \quad\text{e}\quad
        B_A[(1,0),1].
    \]
    Em seguida, descreva geometricamente a diferença entre esses dois conjuntos.
\end{exercise}

\begin{exercise}
    Considere
    \[
        A=\mathbb R\times[0,+\infty[\subseteq\mathbb R^2
    \]
    com a métrica herdada da distância euclidiana usual de $\mathbb R^2$.
    Seja $p=(0,1)$.
    Descreva geometricamente a bola aberta
    \[
        B_A(p,1/2).
    \]
    Ela coincide com uma bola aberta de $\mathbb R^2$?
\end{exercise}

\begin{exercise}
    Seja $M$ um conjunto.
    Defina $d:M\times M\to\mathbb R$ por
    \[
        d(p,q)=
        \begin{cases}
            0, & \text{se } p=q,\\
            1, & \text{se } p\neq q.
        \end{cases}
    \]
    Mostre que $d$ é uma métrica em $M$.
    Essa métrica é chamada de \emph{métrica discreta}.
\end{exercise}

\begin{exercise}
    Seja $M$ um conjunto munido da métrica discreta.
    Fixe $p\in M$.
    Determine $B(p,r)$ e $B[p,r]$ nos seguintes casos:
    \[
        0<r<1,\qquad r=1,\qquad r>1.
    \]
\end{exercise}

\begin{exercise}
    Considere $\mathbb Z$ como subespaço de $\mathbb R$, com a métrica herdada da distância euclidiana usual de $\mathbb R$.
    Dado $n \in \mathbb Z$, determine $B_{\mathbb Z}(n,\frac{1}{2})$ e $B_{\mathbb Z}[n,\frac{1}{2}]$.
\end{exercise}
