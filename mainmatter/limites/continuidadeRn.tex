
\section{Continuidade em \texorpdfstring{$\mathbb R^n$}{Rn}}
Como acontece no caso unidimensional, intuitivamente, uma função $f$ é \emph{contínua}\index{continuidade}\index{função contínua|see{continuidade}} em um ponto $p$ de seu domínio se, e somente se $f(x)$ fica tão próximo de $f(p)$ quanto se queira, desde que $x$ esteja suficientemente próximo de $p$.

Formalmente, temos a definição abaixo.
Lembremos que se $p\in \mathbb R^n$ e $\epsilon>0$, então $B(p, \epsilon)=\{x\in \mathbb R^n: d(x, p)<\epsilon\}$ é a bola aberta de centro $p$ e raio $\epsilon$.

Ao longo dessa seção, $n$ e $m$ denotarão números inteiros positivos.

\begin{definition}\index{continuidade!definição}
    Sejam:
    \begin{itemize}
        \item $n,m\geq 1$,
        \item $A\subseteq \mathbb R^n$ um conjunto,
        \item $f: A \to \mathbb R^m$ uma função, e
        \item $p\in A$.
    \end{itemize}

    Dizemos que $f$ é \emph{contínua em $p$} se, para todo $\varepsilon > 0$, existe $\delta > 0$ tal que, para todo $x \in A$,
    \[
        \text{se } d(x, p) < \delta, \text{ então } d(f(x), f(p)) < \varepsilon.
    \]

    Dizemos que $f$ é \emph{contínua} se, para todo $q \in A$, $f$ é contínua em $q$.
\end{definition}

Notemos que, se $n=m=1$, temos exatamente a definição de continuidade pra funções de uma variável real a valores reais.

Como um primeiro exemplo, tratemos das projeções.

\begin{definition}\index{projeção}
    Seja $n\geq 1$ e $i$ um inteiro com $1\leq i\leq n$.
    
    A \emph{$i$-ésima projeção} de $\mathbb R^n$ em $\mathbb R$ é a função $\pi_i: \mathbb R^n\rightarrow \mathbb R$ dada por
    \[
        \pi_i(x_1, \dots, x_n) = x_i.
    \]
\end{definition}

\begin{example}
    Para o caso $n=2$, as projeções $\pi_1$ e $\pi_2$ são dadas por $\pi_1(x, y) = x$ e $\pi_2(x, y) = y$.
\end{example}

As projeções são funções contínuas.
\begin{proposition}\index{projeção}
    Seja $n\geq 1$ e $i$ um inteiro com $1\leq i\leq n$.
    
    A $i$-ésima projeção de $\mathbb R^n$ em $\mathbb R$, ou seja, a função é contínua.
    
    Ou seja, seja $\pi_i: \mathbb R^n\rightarrow \mathbb R$ dada por $\pi_i(x_1, \dots, x_n) = x_i$.

    Então $\pi_i$ é contínua.
\end{proposition}
\begin{proof}
    Seja $p=(p_1, \dots, p_n) \in \mathbb R^n$ arbitrário e $\epsilon>0$.

    Note que, dado qualquer $x=(x_1, \dots, x_n) \in \mathbb R^n$, temos que:
    \begin{equation*}
        d(\pi_i(x), \pi_i(p)) = |x_i - p_i| \leq \sqrt{(x_1 - p_1)^2 + \cdots + (x_n - p_n)^2} = d(x, p).
    \end{equation*}

    Assim, se $d(x, p)<\epsilon$, então $d(\pi_i(x), \pi_i(p))<\epsilon$.
    Logo, tomando $\delta=\epsilon$, vemos que $\pi_i$ é contínua em $p$.

    Como $p\in \mathbb R^n$ foi arbitrário, concluímos que $\pi_i$ é contínua.
\end{proof}

Assim, por exemplo, temos que:

\begin{example}
    Seja $f: \mathbb R^2\rightarrow \mathbb R$ dada por $f(x, y) = x$.
    Então $f$ é contínua.
\end{example}

\begin{example}
    Seja $f: \mathbb R\rightarrow \mathbb R$ dada por $f(x) = x$.
    Então $f$ é contínua, pois $f$ é a projeção $\pi_1$ de $\mathbb R^1$ em $\mathbb R$.
\end{example}

\begin{example}
    Seja $f: \mathbb R^3\rightarrow \mathbb R$ dada por $f(u, v, w)=v$.
    Então $f$ é contínua, pois $f$ é a projeção $\pi_2$ de $\mathbb R^3$ em $\mathbb R$.
\end{example}

Outra classe de exemplos de funções contínuas simples são as funções constantes, ou seja, as funções da forma $f(x_1, \dots, x_n) = c$ para algum $c \in \mathbb R^m$.

\begin{proposition}\index{função constante}
    Sejam:
    \begin{itemize}
        \item $n,m\geq 1$
        \item $c \in \mathbb R^m$.
    \end{itemize}

    Seja $f: \mathbb R^n\rightarrow \mathbb R^m$ uma função tal que, para todo $x \in \mathbb R^n$,
    \[
        f(x) = c.
    \]

    Então $f$ é contínua.
\end{proposition}
\begin{proof}
    Seja $p=(p_1, \dots, p_n) \in \mathbb R^n$ arbitrário e $\epsilon>0$.

    Note que, dado qualquer $x=(x_1, \dots, x_n) \in \mathbb R^n$, temos que:
    \begin{equation*}
        d(f(x), f(p)) = d(c, c) = 0 < \epsilon.
    \end{equation*}

    Assim, se $d(x, p)<\epsilon$, então $d(f(x), f(p))<\epsilon$.
    Logo, tomando $\delta=\epsilon$, vemos que $f$ é contínua em $p$.

    Como $p\in \mathbb R^n$ foi arbitrário, concluímos que $f$ é contínua.
\end{proof}
