\section{O Teorema do Confronto}\index{Teorema do Confronto}\index{sanduíche, teorema do|see{Teorema do Confronto}}

Às vezes, ao se calcular o limite de uma função complicada, basta aprisioná-la entre duas funções cujo comportamento já conhecemos que possuem um mesmo limite.

Essa é a ideia é formalizada pelo Teorema do Confronto.

\begin{proposition}[Teorema do Confronto]\index{limite!Teorema do Confronto}
    Sejam:
    \begin{itemize}
        \item $(M,d_M)$ um espaço métrico,
        \item $A\subseteq M$ um subespaço,
        \item $f,g,h:A\to\mathbb R$ funções,
        \item $p\in M$ um ponto de acumulação de $A$, e
        \item $L\in\mathbb R$.
    \end{itemize}

    Suponha que existe $\rho>0$ tal que, para todo $x\in A\setminus\{p\}$,
    \[
        d_M(x,p)<\rho
        \quad\Longrightarrow\quad
        f(x)\leq g(x)\leq h(x).
    \]

    Se
    \[
        \lim_{x\to p}f(x)=L
        \qquad\text{e}\qquad
        \lim_{x\to p}h(x)=L,
    \]
    então
    \[
        \lim_{x\to p}g(x)=L.
    \]
\end{proposition}

\begin{proof}
    Fixe $\epsilon>0$.

    Como $\lim_{x\to p}f(x)=L$, existe $\delta_f>0$ tal que, para todo $x\in A\setminus\{p\}$,
    \[
        \text{se }d_M(x,p)<\delta_f,\text{ então } |f(x)-L|<\epsilon.
    \]
    Em particular, para tais $x$,
    \[
        L-\epsilon<f(x).
    \]

    Como $\lim_{x\to p}h(x)=L$, existe $\delta_h>0$ tal que, para todo $x\in A\setminus\{p\}$,
    \[
        \text{se }d_M(x,p)<\delta_h,\text{ então } |h(x)-L|<\epsilon.
    \]
    Em particular, para tais $x$,
    \[
        h(x)<L+\epsilon.
    \]

    Tome
    \[
        \delta=\min\{\rho,\delta_f,\delta_h\}.
    \]

    Se $x\in A\setminus\{p\}$ e $d_M(x,p)<\delta$, então
    \[
        L-\epsilon<f(x)\leq g(x)\leq h(x)<L+\epsilon.
    \]
    Logo,
    \[
        \epsilon<g(x)-L<\epsilon,
    \]
    e, portanto,
    \[
        |g(x)-L|<\epsilon.
    \]

    Como $\epsilon>0$ foi arbitrário, concluímos que $\lim_{x\to p}g(x)=L$.
\end{proof}

\begin{corollary}\index{limite!dominação por uma função que tende a zero}
    Sejam:
    \begin{itemize}
        \item $(M,d_M)$ um espaço métrico,
        \item $A\subseteq M$ um subespaço,
        \item $u:A\to\mathbb R$ e $v:A\to\mathbb R$ funções,
        \item $p\in M$ um ponto de acumulação de $A$, e
        \item $L\in\mathbb R$.
    \end{itemize}

    Suponha que existe $\rho>0$ tal que, para todo $x\in A\setminus\{p\}$,
    \[
        d_M(x,p)<\rho
        \quad\Longrightarrow\quad
        |u(x)-L|\leq v(x).
    \]

    Se
    \[
        \lim_{x\to p}v(x)=0,
    \]
    então
    \[
        \lim_{x\to p}u(x)=L.
    \]
\end{corollary}

\begin{proof}
    Da desigualdade $|u(x)-L|\leq v(x)$, obtemos
    \[
        -v(x)\leq u(x)-L\leq v(x).
    \]
    Somando $L$ em todos os termos,
    \[
        L-v(x)\leq u(x)\leq L+v(x).
    \]

    Pelas regras de operações com limites,
    \[
        \lim_{x\to p}(L-v(x))=L
        \qquad\text{e}\qquad
        \lim_{x\to p}(L+v(x))=L.
    \]

    O Teorema do Confronto implica, então, que
    \[
        \lim_{x\to p}u(x)=L.
    \]
\end{proof}

\begin{corollary}
    Sejam:
    \begin{itemize}
        \item $(M,d_M)$ um espaço métrico,
        \item $A\subseteq M$ um subespaço,
        \item $f,g:A\to\mathbb R$ funções, e
        \item $p\in M$ um ponto de acumulação de $A$.
    \end{itemize}

    Suponha que:
    \begin{itemize}
        \item $\lim_{x\to p}f(x)=0$, e
        \item $g$ é limitada em uma vizinhança perfurada de $p$, isto é, existem $\rho>0$ e $C>0$ tais que, para todo $x\in A\setminus\{p\}$,
        \[
            d_M(x,p)<\rho
            \quad\Longrightarrow\quad
            |g(x)|\leq C.
        \]
    \end{itemize}

    Então
    \[
        \lim_{x\to p}f(x)g(x)=0.
    \]
\end{corollary}

\begin{proof}
    Para todo $x\in A\setminus\{p\}$ suficientemente próximo de $p$, temos
    \[
        |f(x)g(x)|\leq C|f(x)|.
    \]

    Como $\lim_{x\to p}f(x)=0$, temos também
    \[
        \lim_{x\to p}C|f(x)|=0.
    \]
    De fato, dado $\epsilon>0$, basta tomar $x$ suficientemente próximo de $p$ para que $|f(x)|<\frac{\epsilon}{C}$.

    Pelo corolário anterior, aplicado a $u=fg$, $v=C|f|$ e $L=0$, segue que
    \[
        \lim_{x\to p}f(x)g(x)=0.
    \]
\end{proof}

\begin{example}
    Consideremos a função $F:\mathbb R^2\setminus\{(0,0)\}\to\mathbb R$ dada por
    \[
        F(x,y)=\frac{x^2y}{x^2+y^2}.
    \]

    Vamos calcular
    \[
        \lim_{(x,y)\to(0,0)}F(x,y).
    \]

    Para $(x,y)\neq(0,0)$, temos $x^2\leq x^2+y^2$.
    Portanto,
    \[
        |F(x,y)|
        =
        \left|\frac{x^2y}{x^2+y^2}\right|
        =
        \frac{x^2|y|}{x^2+y^2}
        \leq
        |y|.
    \]

    Além disso,
    \[
        |y|\leq \sqrt{x^2+y^2}.
    \]
    Como $\sqrt{x^2+y^2}=d((x,y),(0,0))$ tende a $0$ quando $(x,y)$ tende a $(0,0)$, o Teorema do Confronto dá
    \[
        \lim_{(x,y)\to(0,0)}F(x,y)=0.
    \]
\end{example}

\begin{example}
    Seja $G:\mathbb R^2\setminus\{(0,0)\}\to\mathbb R$ dada por
    \[
        G(x,y)=x^2\sin\left(\frac{1}{x^2+y^2}\right).
    \]

    Como $|\sin t|\leq 1$ para todo $t\in\mathbb R$, temos
    \[
        |G(x,y)|
        \leq
        x^2
    \]

    Como $\lim_{(x,y)\to(0,0)}x^2=0$, segue do Teorema do Confronto que
    \[
        \lim_{(x,y)\to(0,0)}G(x,y)=0.
    \]

    Assim, a função $\widetilde G:\mathbb R^2\to\mathbb R$ dada por
    \[
        \widetilde G(x,y)=
        \begin{cases}
            x^2\sin\left(\frac{1}{x^2+y^2}\right), & \text{se }(x,y)\neq(0,0),\\
            0, & \text{se }(x,y)=(0,0),
        \end{cases}
    \]
    é contínua em $(0,0)$.
\end{example}

\begin{example}
    O Teorema do Confronto também é útil para reconhecer limites de funções que oscilam muito.
    Por exemplo, a função
    \[
        H(x,y)=\|(x,y)\|\cos\left(\frac{1}{\|(x,y)\|}\right),
        \qquad (x,y)\neq(0,0),
    \]
    satisfaz
    \[
        |H(x,y)|\leq \|(x,y)\|.
    \]
    Como $\|(x,y)\|\to 0$ quando $(x,y)\to(0,0)$, concluímos que
    \[
        \lim_{(x,y)\to(0,0)}H(x,y)=0.
    \]
\end{example}

\section*{Exercícios}

\begin{exercise}
    Use o Teorema do Confronto para calcular
    \[
        \lim_{(x,y)\to(0,0)}\frac{xy^2}{x^2+y^2}.
    \]
\end{exercise}

\begin{exercise}
    Mostre que a função $f:\mathbb R^2\to\mathbb R$ dada por
    \[
        f(x,y)=
        \begin{cases}
            \dfrac{x^3}{x^2+y^2}, & \text{se }(x,y)\neq(0,0),\\
            0, & \text{se }(x,y)=(0,0),
        \end{cases}
    \]
    é contínua em $(0,0)$.
\end{exercise}

\begin{exercise}
    Calcule, se existir,
    \[
        \lim_{(x,y)\to(0,0)}(x^2+y^2)\sin\left(\frac{x^2}{x^2+y^2}\right).
    \]
\end{exercise}
